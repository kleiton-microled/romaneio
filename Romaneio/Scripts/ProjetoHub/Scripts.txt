CREATE FUNCTION [dbo].[FN_ROMANEIO_VERIFICAR_CUBAGEM] (@AUTONUM_ROMANEIO INT)
RETURNS BIT
AS
BEGIN

    DECLARE @CubagemCalculada DECIMAL(18,6)
    DECLARE @CubagemPrevista DECIMAL(18,6)
    DECLARE @PercPermitido DECIMAL(10,2)
    DECLARE @PercDiferenca DECIMAL(10,2)
    DECLARE @Resultado BIT = 0


	-- cubagem calculada pelas CLA
    SELECT 
        @CubagemCalculada = SUM(
    QUANTIDADE *
    CAST(COMPRIMENTO AS DECIMAL(18,3)) *
    CAST(LARGURA AS DECIMAL(18,3)) *
    CAST(ALTURA AS DECIMAL(18,3))
)   
    FROM TB_ROMANEIO_BREAK_BULK
    WHERE AUTONUM_ROMANEIO  = @AUTONUM_ROMANEIO


    -- cubagem prevista
    SELECT 
        @CubagemPrevista = VOLUME_M3
    FROM TB_ROMANEIO 
    WHERE AUTONUM_ROMANEIO  = @AUTONUM_ROMANEIO
 


    -- percentual permitido
    SELECT 
        @PercPermitido = MAX_DIVERGENCIA_CUBAGEM
    FROM TB_PARAMETROS_SISTEMA


    -- cálculo da diferença percentual
    SET @PercDiferenca = 
        ((@CubagemCalculada - @CubagemPrevista) / @CubagemPrevista) * 100


    -- verifica se ultrapassou o permitido (para mais ou para menos)
    IF ABS(@PercDiferenca) > @PercPermitido
        SET @Resultado = 1
    ELSE
        SET @Resultado = 0

    RETURN @Resultado


END
GO


CREATE FUNCTION [dbo].[FN_VERIFICAR_CUBAGEM_CS] (@AUTONUM_CS INT)
RETURNS BIT
AS
BEGIN

    DECLARE @CubagemCalculada DECIMAL(18,6)
    DECLARE @CubagemPrevista DECIMAL(18,6)
    DECLARE @PercPermitido DECIMAL(10,2)
    DECLARE @PercDiferenca DECIMAL(10,2)
    DECLARE @Resultado BIT = 0


    -- cubagem calculada pelas CLA
    SELECT @CubagemCalculada = SUM(QUANTIDADE * COMPRIMENTO * LARGURA * ALTURA)
    FROM TB_BREAK_BULK
    WHERE AUTONUM_CS = @AUTONUM_CS


    -- cubagem prevista
    SELECT 
        @CubagemPrevista = VOLUME --VOLUME_M3
    FROM TB_CARGA_SOLTA
    WHERE AUTONUM = @AUTONUM_CS



    -- percentual permitido
    SELECT 
        @PercPermitido = MAX_DIVERGENCIA_CUBAGEM
    FROM TB_PARAMETROS_SISTEMA


    -- cálculo da diferença percentual
    SET @PercDiferenca = 
        ((@CubagemCalculada - @CubagemPrevista) / @CubagemPrevista) * 100


    -- verifica se ultrapassou o permitido (para mais ou para menos)
    IF ABS(@PercDiferenca) > @PercPermitido
        SET @Resultado = 1
    ELSE
        SET @Resultado = 0

    RETURN @Resultado

END
GO
