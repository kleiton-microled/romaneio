ALTER VIEW [dbo].[VW_ROMANEIO_FECHAMENTO_LOTE_ITENS]
AS
 
SELECT DISTINCT  R.AUTONUM_ROMANEIO,
                         A.AUTONUM as AUTONUM_CNTR ,
						 A.ID_CONTEINER,
						 C.AUTONUM AS LOTE, C.NUMERO AS BL, 
						 E.ITEM AS ITEM,
						 E.MARCA AS MARCA ,
						 E.QUANTIDADE AS QUANTIDADE ,
						 E.NCM AS NCM ,
						 E.MERCADORIA AS MERCADORIA ,
						 E.GENERO AS GENERO ,
						 E.VOLUME_M3 AS VOLUME ,
						 E.PESO_BRUTO AS PESO_BRUTO ,
						 E.PESO_REAL AS PESO_APURADO ,
						 E.EMBALAGEM AS EMBALAGEM ,
						 E.IMO AS IMO ,
						 E.UNDG AS UNDG ,
						 T.LOCAL AS LOCAL,
						 T.DESCR AS SUB_CLASSE,  
                         F.RAZAO AS IMPORTADOR, G.DESCR AS SHIPPER, 
						  A.DESCRICAO_SIGVIG ,A.YARD AS POSICAO, 
						  C.FLAG_ACRESCIMO  AS FL_ACRESCIMO,
						   A.PATIO,  
						   CONVERT(VARCHAR,A.PATIO) + ' - '  + CONVERT(VARCHAR,M.NOME_FANTASIA) AS EMPRESA, P.DESCR TIPO_CARGA,
						    ISNULL(MERCADORIADESC1,'') + ISNULL(MERCADORIADESC2,'') AS DESCRICAO,	   T.local AS LOCALIZACAO  

FROM            dbo.TB_CNTR_BL AS A INNER JOIN
                         dbo.TB_AMR_CNTR_BL AS B ON B.CNTR = A.AUTONUM INNER JOIN
						 -- vai ter varios bl no cntr
                         dbo.TB_BL AS C ON C.AUTONUM = B.BL LEFT JOIN
						 SGIPA..TB_DESCONSOLIDACAO_CE S ON S.CE = C.CE
						 INNER JOIN
                         dbo.DTE_TB_TIPOS_CONTEINER AS D ON D.CODE = A.TIPO LEFT JOIN
						 -- vai ter varios itens no bl / cntr
                         dbo.TB_CARGA_CNTR AS E ON C.AUTONUM = E.BL AND E.ID_CONTEINER=A.ID_CONTEINER LEFT JOIN
                         dbo.TB_CAD_PARCEIROS AS F ON F.AUTONUM = C.IMPORTADOR LEFT JOIN
                         dbo.DTE_TB_ARMADORES AS G ON C.ARMADOR = G.CODE LEFT JOIN
                         dbo.TB_TIPO_CARGA_BAND AS P ON C.TIPO_CARGA =P.AUTONUM   LEFT JOIN
                         dbo.TB_TIPO_SUB_CLASSE AS T ON T.AUTONUM = C.SUB_CLASSE INNER JOIN
						dbo.TB_ROMANEIO R ON R.LOTE =C.AUTONUM AND R.AUTONUM_CNTR = A.AUTONUM LEFT JOIN
						dbo.TB_EMPRESAS M ON A.PATIO = M.AUTONUM  
 WHERE    (A.DT_SAIDA IS NULL) AND
 (A.FLAG_TERMINAL = 1) 
	 AND ISNULL((SELECT DISTINCT COUNT(*)QTD FROM TB_ROMANEIO VW WHERE  VW.AUTONUM_CNTR = R.AUTONUM_CNTR AND VW.TERMINO IS NULL),0) = 0
  GROUP BY A.AUTONUM, A.ID_CONTEINER, D.DESCR , A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO , A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM, A.BOL_DESOVA_NUM, A.AUTONUM_SOLICITOU_DESOVA , C.AUTONUM , C.NUMERO, T.LOCAL, F.RAZAO  , G.DESCR  , A.DESCRICAO_SIGVIG , A.YARD , C.FLAG_ACRESCIMO ,A.PATIO,A.REGIME ,T.LOCAL, T.DESCR, P.DESCR, A.BOL_DESOVA_ANO, E.ITEM, E.MARCA, E.QUANTIDADE, E.NCM, E.MERCADORIA, E.GENERO, A.YARD , C.FLAG_ACRESCIMO  , E.VOLUME_M3, E.PESO_BRUTO, E.PESO_REAL , E.EMBALAGEM, E.IMO, E.UNDG, MERCADORIADESC1, MERCADORIADESC2,M.NOME_FANTASIA,  R.AUTONUM_ROMANEIO
 
GO





ALTER VIEW [dbo].[VW_ROMANEIO_FECHAMENTO_LOTE_CNTR]
AS
 

SELECT DISTINCT 
                           A.AUTONUM as AUTONUM_CNTR , A.ID_CONTEINER, D.DESCR AS DESCR_TIPO, A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO AS DT_AGENDAMENTO, A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM,
						   CONVERT(VARCHAR,A.BOL_DESOVA_NUM) + '/' + CONVERT(VARCHAR,A.BOL_DESOVA_ANO) AS BOL_DESUNIT, 
						 CASE WHEN A.REGIME ='PP' THEN 'NÃO' ELSE 'SIM ' END  DDA, 
                         A.AUTONUM_SOLICITOU_DESOVA AS SOLICITADO, 
 						  R.INICIO  AS ABERTURA, 
						  R.TERMINO  AS FECHAMENTO, 
						 C.AUTONUM AS LOTE, C.NUMERO AS BL, 
						 COUNT(E.ITEM) AS ITEM,
						 STRING_AGG(E.MARCA, ', ') AS MARCA ,
						 SUM(ISNULL(E.QUANTIDADE,0))AS QUANTIDADE ,
						 STRING_AGG(E.NCM, ', ') AS NCM ,
						 STRING_AGG(E.MERCADORIA, ', ') AS MERCADORIA ,
						 STRING_AGG(E.GENERO, ', ') AS GENERO ,
						 STRING_AGG(E.VOLUME_M3, ', ') AS VOLUME ,
						 STRING_AGG(E.PESO_BRUTO, ', ') AS PESO_BRUTO ,
						 STRING_AGG(E.PESO_REAL, ', ') AS PESO_APURADO ,
						 STRING_AGG(E.EMBALAGEM, ', ') AS EMBALAGEM ,
						 STRING_AGG(E.IMO, ', ') AS IMO ,
						 STRING_AGG(E.UNDG, ', ') AS UNDG ,
						  T.local AS LOCALIZACAO,  
                         F.RAZAO AS IMPORTADOR, G.DESCR AS SHIPPER, 
						  A.DESCRICAO_SIGVIG ,A.YARD AS POSICAO, 
						  C.FLAG_ACRESCIMO  AS FL_ACRESCIMO,
						   A.PATIO, TC.DESCR TIPO_CARGA
FROM            dbo.TB_CNTR_BL AS A INNER JOIN
                         dbo.TB_AMR_CNTR_BL AS B ON B.CNTR = A.AUTONUM INNER JOIN
						 -- vai ter varios bl no cntr
                         dbo.TB_BL AS C ON C.AUTONUM = B.BL AND C.AUTONUM = B.BL INNER JOIN
                         dbo.DTE_TB_TIPOS_CONTEINER AS D ON D.CODE = A.TIPO LEFT JOIN
						 -- vai ter varios itens no bl / cntr
                         dbo.TB_CARGA_CNTR AS E ON C.AUTONUM = E.BL AND E.ID_CONTEINER=A.ID_CONTEINER LEFT JOIN
                         dbo.TB_CAD_PARCEIROS AS F ON F.AUTONUM = C.IMPORTADOR LEFT JOIN
                         dbo.DTE_TB_ARMADORES AS G ON C.ARMADOR = G.CODE LEFT JOIN
                         dbo.TB_TIPO_CARGA_BAND AS TC ON C.TIPO_CARGA =TC.AUTONUM   LEFT JOIN
                         dbo.TB_TIPO_SUB_CLASSE AS T ON T.AUTONUM = C.SUB_CLASSE INNER JOIN
						dbo.TB_ROMANEIO R ON R.LOTE =C.AUTONUM AND R.AUTONUM_CNTR = A.AUTONUM 
 WHERE   (A.DT_SAIDA IS NULL) AND
 (A.FLAG_TERMINAL = 1) 
     AND A.REGIME IN('PP','HP') 
	 AND ISNULL((SELECT DISTINCT COUNT(*)QTD FROM TB_ROMANEIO VW WHERE  VW.AUTONUM_CNTR = R.AUTONUM_CNTR AND VW.TERMINO IS NULL),0) = 0
  GROUP BY A.AUTONUM, A.ID_CONTEINER, D.DESCR , A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO , A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM, A.BOL_DESOVA_NUM, A.AUTONUM_SOLICITOU_DESOVA, A.DESCRICAO_SIGVIG , A.YARD  ,A.PATIO,A.REGIME  , C.AUTONUM , C.NUMERO, C.FLAG_ACRESCIMO, T.LOCAL, F.RAZAO  , G.DESCR  , TC.DESCR, A.BOL_DESOVA_ANO,R.INICIO,R.TERMINO
 
GO

FROM TB_ROMANEIO_AVARIAS A
INNER JOIN INSERTED RO ON RO.AUTONUM_ROMANEIO = A.AUTONUM_ROMANEIO

INNER JOIN INSERTED RO ON RO.AUTONUM_ROMANEIO = A.AUTONUM_ROMANEIO
WHERE A.AUTONUM_ROMANEIO_AVARIAS =  @AUTONUM_ROMANEIO_AVARIAS

    FETCH NEXT FROM TB_ROMANEIO_AVARIAS INTO @AUTONUM_ROMANEIO_AVARIAS
END

 WHERE RO.LOTE = BL.AUTONUM

 FROM SGIPA..TB_CARGA_CNTR TAB 
 INNER JOIN INSERTED RO ON RO.LOTE = TAB.BL AND RO.ITEM= TAB.ITEM 
 INNER JOIN SGIPA..TB_CNTR_BL CNTR ON RO.AUTONUM_CNTR = CNTR.AUTONUM AND TAB.id_conteiner = CNTR.ID_CONTEINER
 WHERE  RO.LOTE = TAB.BL AND RO.ITEM= TAB.ITEM 
 AND RO.AUTONUM_CNTR = CNTR.AUTONUM AND TAB.id_conteiner = CNTR.ID_CONTEINER

     END
 END
