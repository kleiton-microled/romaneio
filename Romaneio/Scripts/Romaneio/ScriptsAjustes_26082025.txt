ALTER VIEW [dbo].[VW_ROMANEIO_CONSULTA_LOTE_CNTR]
AS
 SELECT DISTINCT 
                         A.AUTONUM as AUTONUM_CNTR , A.ID_CONTEINER, D.DESCR AS DESCR_TIPO, A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO AS DT_AGENDAMENTO, A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM,
						 CONVERT(VARCHAR,A.BOL_DESOVA_NUM) + '/' + CONVERT(VARCHAR,A.BOL_DESOVA_ANO) AS BOL_DESUNIT, 
						 CASE WHEN A.REGIME ='PP' THEN 'NÃO' ELSE 'SIM ' END  DDA, 
                         A.AUTONUM_SOLICITOU_DESOVA AS SOLICITADO, 
 						 R.INICIO  AS ABERTURA, 
						 R.TERMINO  AS FECHAMENTO, 
						 C.AUTONUM AS LOTE, C.NUMERO AS BL, 
						 COUNT(E.ITEM) AS ITEM,
						 STRING_AGG(E.MARCA, ', ') AS MARCA ,
						 SUM(ISNULL(E.QUANTIDADE,0))AS QUANTIDADE ,
						 STRING_AGG(E.NCM, ', ') AS NCM ,
						 STRING_AGG(E.MERCADORIA, ', ') AS MERCADORIA ,
						 STRING_AGG(E.GENERO, ', ') AS GENERO ,
						 STRING_AGG(E.VOLUME_M3, ', ') AS VOLUME ,
						 STRING_AGG(E.PESO_BRUTO, ', ') AS PESO_BRUTO ,
						 STRING_AGG(E.PESO_REAL, ', ') AS PESO_APURADO ,
						 STRING_AGG(E.EMBALAGEM, ', ') AS EMBALAGEM ,
						 STRING_AGG(E.IMO, ', ') AS IMO ,
						 STRING_AGG(E.UNDG, ', ') AS UNDG ,
						 T.local AS LOCALIZACAO,  
                         F.RAZAO AS IMPORTADOR, G.DESCR AS SHIPPER, 
						 A.DESCRICAO_SIGVIG ,A.YARD AS POSICAO, 
						 C.FLAG_ACRESCIMO  AS FL_ACRESCIMO,
						 A.PATIO, TC.DESCR TIPO_CARGA
FROM            dbo.TB_CNTR_BL AS A INNER JOIN
                         dbo.TB_AMR_CNTR_BL AS B ON B.CNTR = A.AUTONUM INNER JOIN
						 -- vai ter varios bl no cntr
                         dbo.TB_BL AS C ON C.AUTONUM = B.BL AND C.AUTONUM = B.BL INNER JOIN
                         dbo.DTE_TB_TIPOS_CONTEINER AS D ON D.CODE = A.TIPO LEFT JOIN
						 -- vai ter varios itens no bl / cntr
                         dbo.TB_CARGA_CNTR AS E ON C.AUTONUM = E.BL AND E.ID_CONTEINER=A.ID_CONTEINER LEFT JOIN
                         dbo.TB_CAD_PARCEIROS AS F ON F.AUTONUM = C.IMPORTADOR LEFT JOIN
                         dbo.DTE_TB_ARMADORES AS G ON C.ARMADOR = G.CODE LEFT JOIN
                         dbo.TB_TIPO_CARGA_BAND AS TC ON C.TIPO_CARGA =TC.AUTONUM   LEFT JOIN
                         dbo.TB_TIPO_SUB_CLASSE AS T ON T.AUTONUM = C.SUB_CLASSE LEFT JOIN
						dbo.TB_ROMANEIO R ON R.LOTE =C.AUTONUM AND R.AUTONUM_CNTR = A.AUTONUM
 WHERE      (A.DT_SAIDA IS NULL) AND (A.FLAG_TERMINAL = 1) AND C.FLAG_ATIVO = 1 
    AND ISNULL(flag_desovado,0)=0 AND A.REGIME IN('PP','HP') 
  GROUP BY A.AUTONUM, A.ID_CONTEINER, D.DESCR , A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO , A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM, A.BOL_DESOVA_NUM, A.AUTONUM_SOLICITOU_DESOVA, A.DESCRICAO_SIGVIG , A.YARD  ,A.PATIO,A.REGIME  , C.AUTONUM , C.NUMERO, C.FLAG_ACRESCIMO, T.LOCAL, F.RAZAO  , G.DESCR  , TC.DESCR, A.BOL_DESOVA_ANO,R.INICIO,R.TERMINO
 
 
GO
