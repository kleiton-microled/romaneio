CREATE VIEW [dbo].[VW_ROMANEIO_CONSULTA_LOTE_ITENS]
AS
 
SELECT DISTINCT 
                         A.AUTONUM as AUTONUM_CNTR ,
						 A.ID_CONTEINER,						 
						 C.AUTONUM AS LOTE, C.NUMERO AS BL, 
						 E.ITEM AS ITEM,
						 E.MARCA AS MARCA ,
						 E.QUANTIDADE AS QUANTIDADE ,
						 E.NCM AS NCM ,
						 E.MERCADORIA AS MERCADORIA ,
						 E.GENERO AS GENERO ,
						 E.VOLUME_M3 AS VOLUME ,
						 E.PESO_BRUTO AS PESO_BRUTO ,
						 E.PESO_REAL AS PESO_APURADO ,
						 E.EMBALAGEM AS EMBALAGEM ,
						 E.IMO AS IMO ,
						 E.UNDG AS UNDG ,
						 T.LOCAL AS LOCAL,
						 T.DESCR AS SUB_CLASSE,  
                         F.RAZAO AS IMPORTADOR, G.DESCR AS SHIPPER, 
						  A.DESCRICAO_SIGVIG ,A.YARD AS POSICAO, 
						  C.FLAG_ACRESCIMO  AS FL_ACRESCIMO,
						   A.PATIO,  
						   CONVERT(VARCHAR,A.PATIO) + ' - '  + CONVERT(VARCHAR,M.NOME_FANTASIA) AS EMPRESA, P.DESCR TIPO_CARGA,
						    ISNULL(MERCADORIADESC1,'') + ISNULL(MERCADORIADESC2,'') AS DESCRICAO
FROM            dbo.TB_CNTR_BL AS A INNER JOIN
                         dbo.TB_AMR_CNTR_BL AS B ON B.CNTR = A.AUTONUM INNER JOIN
						 -- vai ter varios bl no cntr
                         dbo.TB_BL AS C ON C.AUTONUM = B.BL LEFT JOIN
						 SGIPA..TB_DESCONSOLIDACAO_CE S ON S.CE = C.CE
						 INNER JOIN
                         dbo.DTE_TB_TIPOS_CONTEINER AS D ON D.CODE = A.TIPO LEFT JOIN
						 -- vai ter varios itens no bl / cntr
                         dbo.TB_CARGA_CNTR AS E ON C.AUTONUM = E.BL AND E.ID_CONTEINER=A.ID_CONTEINER LEFT JOIN
                         dbo.TB_CAD_PARCEIROS AS F ON F.AUTONUM = C.IMPORTADOR LEFT JOIN
                         dbo.DTE_TB_ARMADORES AS G ON C.ARMADOR = G.CODE LEFT JOIN
                         dbo.TB_TIPO_CARGA_BAND AS P ON C.TIPO_CARGA =P.AUTONUM   LEFT JOIN
                         dbo.TB_TIPO_SUB_CLASSE AS T ON T.AUTONUM = C.SUB_CLASSE LEFT JOIN
						dbo.TB_ROMANEIO R ON R.LOTE =C.AUTONUM AND R.AUTONUM_CNTR = A.AUTONUM LEFT JOIN
						dbo.TB_EMPRESAS M ON A.PATIO = M.AUTONUM  
 WHERE      (A.DT_SAIDA IS NULL) AND (A.FLAG_TERMINAL = 1) 
    and isnull(flag_desovado,0)=0 AND A.REGIME IN('PP','HP') 
  GROUP BY A.AUTONUM, A.ID_CONTEINER, D.DESCR , A.TAMANHO, A.DT_ENTRADA, A.DT_AGENDAMENTO_LIBERACAO , A.LACRE_ORIGEM, A.LACRE_IPA, A.VIAGEM, A.BOL_DESOVA_NUM, A.AUTONUM_SOLICITOU_DESOVA , C.AUTONUM , C.NUMERO, T.LOCAL, F.RAZAO  , G.DESCR  , A.DESCRICAO_SIGVIG , A.YARD , C.FLAG_ACRESCIMO ,A.PATIO,A.REGIME ,T.LOCAL, T.DESCR, P.DESCR, A.BOL_DESOVA_ANO, E.ITEM, E.MARCA, E.QUANTIDADE, E.NCM, E.MERCADORIA, E.GENERO, A.YARD , C.FLAG_ACRESCIMO  , E.VOLUME_M3, E.PESO_BRUTO, E.PESO_REAL , E.EMBALAGEM, E.IMO, E.UNDG, MERCADORIADESC1, MERCADORIADESC2,M.NOME_FANTASIA

 
GO

CREATE VIEW [dbo].[VW_ROMANEIO_CONSULTA]
AS
SELECT DISTINCT 
A.AUTONUM_ROMANEIO, 
A.LOTE, 
A.AUTONUM_CNTR, 
CONVERT(VARCHAR,A.INICIO,103)INICIO, 
CONVERT(VARCHAR,A.DEMURRAGE,103)DEMURRAGE, 
CONVERT(VARCHAR,A.TERMINO,103)TERMINO, 
ISNULL(A.SOLICITANTE,0)SOLICITANTE, 
A.CONFERENTE, 
A.FISCAL, 
A.DESOVA, 
CONVERT(VARCHAR,A.INSPECAO_MAPA,103)INSPECAO_MAPA, 
A.VOLUME_M3 as VOLUME, 
A.PESO_APURADO, 
A.PESO_BRUTO , 
A.IMO, 
A.UNDG, 
ISNULL(A.GENERO,0)GENERO, 
ISNULL(A.ITEM,0)ITEM, 
A.QUANTIDADE, 
ISNULL(A.EMBALAGEM,0)EMBALAGEM, 
A.MERCADORIA, 
A.MARCA, 
A.OBS, 
A.FL_OUTRA_AVARIA, 
A.DESCR_OUTRA_AVARIA, 
A.FL_FALTA, 
A.DESCR_FALTA, 
A.FL_ACRESCIMO, 
A.DESCR_ACRESCIMO, 
A.FL_AGUARDANDO_RECONHECIMENTO, 
A.DESCR_AGUARDANDO_RECONHECIMENTO,
A.LOCALIZACAO,
B.ID_CONTEINER,
T.LOCAL,
T.DESCR AS SUB_CLASSE,
B.PATIO,
CONVERT(VARCHAR,b.PATIO) + ' - '  + CONVERT(VARCHAR,m.NOME_FANTASIA) AS EMPRESA, 
P.DESCR TIPO_CARGA,
ISNULL(s.MERCADORIADESC1,'') + ISNULL(MERCADORIADESC2,'') AS DESCRICAO,
E.NCM AS NCM 
  FROM SGIPA..TB_ROMANEIO A
  LEFT JOIN SGIPA..TB_CNTR_BL B ON A.AUTONUM_CNTR = B.AUTONUM
  LEFT JOIN SGIPA..TB_BL AS C ON A.LOTE = C.AUTONUM  
  LEFT JOIN SGIPA..TB_TIPO_SUB_CLASSE AS T ON T.AUTONUM = C.SUB_CLASSE
  LEFT JOIN SGIPA..TB_TIPO_CARGA_BAND AS P ON C.TIPO_CARGA =P.AUTONUM 
  LEFT JOIN SGIPA..TB_DESCONSOLIDACAO_CE S ON S.CE = C.CE
  LEFT JOIN SGIPA..TB_EMPRESAS M ON b.PATIO = M.AUTONUM 
  LEFT JOIN SGIPA..TB_CARGA_CNTR AS E ON C.AUTONUM = E.BL AND E.ID_CONTEINER=B.ID_CONTEINER
