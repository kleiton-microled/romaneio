ALTER PROCEDURE [dbo].[PROC_ROMANEIO_REASSOCIA_MARCANTES]
    @LOTE INT,
    @ID_CNTR INT,
    @ITEM INT  
WITH EXEC AS CALLER
AS
BEGIN
    DECLARE @MARCANTE INT
    DECLARE @AUTONUMCS INT

    DECLARE TABELA CURSOR FOR
    SELECT 
        C.AUTONUM AS AUTONUMCS, 
        V.AUTONUM AS MARCANTE 
    FROM TB_CARGA_SOLTA C 
    INNER JOIN VW_MARCANTES_REASSOCIAR V 
        ON C.BL = V.BL 
        AND C.CNTR = V.CNTR 
        AND C.ITEM = V.ITEM
    WHERE 
        C.BL = @LOTE
        AND C.CNTR = @ID_CNTR
        AND C.ITEM = @ITEM

    OPEN TABELA
    FETCH NEXT FROM TABELA INTO @AUTONUMCS, @MARCANTE

    WHILE @@FETCH_STATUS = 0
    BEGIN
        UPDATE TB_MARCANTES 
        SET AUTONUM_CARGA = @AUTONUMCS  
        WHERE AUTONUM = @MARCANTE

        FETCH NEXT FROM TABELA INTO @AUTONUMCS, @MARCANTE    
    END 

    CLOSE TABELA  
    DEALLOCATE TABELA  

    -- Segunda parte do código
    DECLARE @AUTONUM INT
    DECLARE @AUTONUM_CARGA INT

    DECLARE TABELA CURSOR FOR
    SELECT 
        Y.AUTONUM,
        M.AUTONUM_CARGA 
    FROM TB_CARGA_SOLTA_YARD Y 
    INNER JOIN TB_MARCANTES M 
        ON Y.AUTONUM = M.AUTONUM_CS_YARD 
    WHERE 
        Y.AUTONUM_CS <> M.AUTONUM_CARGA 
        AND M.AUTONUM_CARGA > 0 
        AND M.BL = @LOTE

    OPEN TABELA
    FETCH NEXT FROM TABELA INTO @AUTONUM, @AUTONUM_CARGA

    WHILE @@FETCH_STATUS = 0
    BEGIN
        UPDATE TB_CARGA_SOLTA_YARD 
        SET AUTONUM_CS = @AUTONUM_CARGA  
        WHERE AUTONUM = @AUTONUM

        FETCH NEXT FROM TABELA INTO @AUTONUM, @AUTONUM_CARGA    
    END 

    CLOSE TABELA  
    DEALLOCATE TABELA  
END
GO


CREATE PROCEDURE [dbo].[PROC_ROMANEIO_ACERTA_DESOVA]
    @LOTE INT,
    @ID_CNTR INT 
 WITH EXEC AS CALLER
AS
BEGIN
    DECLARE @AUTONUM INT
    DECLARE @BL INT
    DECLARE @CNTR INT    
	DECLARE @ITEM INT
	
	DECLARE TABELA CURSOR FOR
   
 SELECT DISTINCT S.AUTONUM,S.BL,S.CNTR,S.ITEM FROM TB_CARGA_SOLTA S 
INNER JOIN TB_MARCANTES M ON
S.CNTR = M.CNTR
AND S.BL=M.BL
AND S.ITEM=M.ITEM
INNER JOIN TB_CNTR_BL C ON S.CNTR=C.AUTONUM
WHERE M.AUTONUM_CARGA = 0 AND (C.DT_FIM_DESOVA IS NOT NULL OR C.FLAG_DESOVADO=1) 
 AND S.BL= @LOTE  AND S.CNTR= @ID_CNTR
 --AND S.DT_SAIDA IS NULL

    OPEN TABELA
    FETCH NEXT FROM TABELA INTO @AUTONUM, @BL, @CNTR, @ITEM;

    WHILE @@FETCH_STATUS = 0
    BEGIN
       UPDATE TB_MARCANTES SET AUTONUM_CARGA=@AUTONUM
	   WHERE AUTONUM_CARGA=0 AND BL=@BL AND CNTR=@CNTR AND ITEM=@ITEM

        FETCH NEXT FROM TABELA INTO @AUTONUM, @BL, @CNTR, @ITEM;
    END 

    CLOSE TABELA  
    DEALLOCATE TABELA  

END
GO
